class Solution {
    public int search(int[] nums, int target) {
        int start = 0, end = nums.length - 1;

        while (start <= end) {
            int mid = start + (end - start) / 2;

            if (nums[mid] == target) return mid;

            // Check if left part is sorted
            if (nums[mid] >= nums[start]) {
                if (target >= nums[start] && target < nums[mid]) {
                    end = mid - 1; // move left
                } else {
                    start = mid + 1; // move right
                }
            } 
            // Right part is sorted
            else {
                if (target > nums[mid] && target <= nums[end]) {
                    start = mid + 1; // move right
                } else {
                    end = mid - 1; // move left
                }
            }
        }

        return -1;
    }
}

using namespace std;
// Build Prefix Array
vector<int> buildPrefixArray(vector<int>& nums) {
    int n = nums.size();
    vector<int> prefix(n);
    prefix[0] = nums[0];
    for (int i = 1; i < n; i++) {
        prefix[i] = prefix[i - 1] + nums[i];
    }
    return prefix;
}
// Build Suffix Array
vector<int> buildSuffixArray(vector<int>& nums) {
    int n = nums.size();
    vector<int> suffix(n);
    suffix[n - 1] = nums[n - 1];
    for (int i = n - 2; i >= 0; i--) {
        suffix[i] = suffix[i + 1] + nums[i];
    }
    return suffix;
}
// Main Logic
void solveProblem(vector<int>& nums) {
    vector<int> prefix = buildPrefixArray(nums);
    vector<int> suffix = buildSuffixArray(nums);
    cout << "Prefix Array: ";
    for (int x : prefix){
        cout << x << " ";
    }
    cout << "\nSuffix Array: ";
    for (int x : suffix){
        cout << x << " ";
    }
}
// Driver Code
int main() {
    vector<int> nums = {1, 2, 3, 4, 5};
    solveProblem(nums);
    return 0;
